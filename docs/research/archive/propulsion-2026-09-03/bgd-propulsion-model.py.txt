def motor(rpm,kv=800,R=.1293,I0=1,cp=1):
 ps=np.interp(rpm,n,p)*cp;q=ps/(rpm*2*np.pi/60);i=q/(60/(2*np.pi*kv))+I0;v=rpm/kv+i*R
 return v,i,ps,v*i

def full(V,kv=800,R=.1293,I0=1,cp=1,ct=1):
 lo,hi=1000,40000
 for _ in range(70):
  x=(lo+hi)/2
  if motor(x,kv,R,I0,cp)[0]>V:hi=x

 return {'rpm':x,'I4':4*i,'thrust_g4':4*np.interp(x,n,t)*ct/9.80665*1000,'shaftW4':ps*4}
