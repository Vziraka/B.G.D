def mot(rpm,kv=950,R=.072,I0=1.2,cp=1):
    ps=np.interp(rpm,nr,shaft)*cp
    q=ps/(rpm*2*np.pi/60)
    i=q/(60/(2*np.pi*kv))+I0
    v=rpm/kv+i*R
    return v,i,ps,v*i

def full(V,kv=950,R=.072,I0=1.2,cp=1,ct=1):
    lo,hi=100,30000
    for _ in range(80):
        x=(lo+hi)/2
        if mot(x,kv,R,I0,cp)[0]>V: hi=x
        else: lo=x
    v,i,ps,pe=mot(x,kv,R,I0,cp)
    return dict(V=V,rpm=x,I4=4*i,thrust_g4=4*np.interp(x,nr,thrustN)*ct/9.80665*1000,shaftW4=ps*4,Pelec4=pe*4)

def hover(massg,kv=950,R=.072,I0=1.2,cp=1,ct=1,k=1.10,aux=18):
    reqN=massg/1000*9.80665/4/ct
    rpm=np.interp(reqN,thrustN,nr)
    v,i,ps,pe=mot(rpm,kv,R,I0,cp)
    P=4*pe*k+aux
    reserve=max(20,.05*P)
    return dict(massg=massg,rpm=rpm,motorV=v,motorI=i,P=P,reserve=reserve)

def energy_to_vcell(target,curve='10',series=5,derate=.95):
    vv=v10 if curve=='10' else v20
    # voltage monotonically falling; find Ah by inverse interp
    target=max(vv[-1],min(vv[0],target))
    aa=np.interp(target,vv[::-1],ah[::-1])
    xs=np.r_[ah[ah<aa],aa]
    ys=np.r_[vv[ah<aa],target]
    E=series*np.trapezoid(ys,xs)
    return aa,E,E*derate

def coupled(vcurve_pack,massg,series=5,baselineI=20,Iaux=4,Rstep=.048,kv=.95*950,R=1.3*.072,I0=1.8,cp=1.15,ct=.9):
    V=vcurve_pack
    for _ in range(1000):
        f=full(V,kv,R,I0,cp,ct)
        new=vcurve_pack-(f['I4']+Iaux-baselineI)*Rstep
        if abs(new-V)<1e-12: break
        V=(V+new)/2
    f=full(V,kv,R,I0,cp,ct)
    f.update(vcurve_pack=vcurve_pack,massg=massg,ratio=f['thrust_g4']/massg,totalA=f['I4']+Iaux)
    return f
